----------------------------------------
РАСПОЛОЖЕНИЕ ЭЛЕМЕНТОВ
----------------------------------------
display: block;                  блочный
display: inline-block;           в строку + размеры
display: none;                   скрыть
display: flex;                   выравнивание в ряд

/* Flex — используется постоянно */
display: flex;
justify-content: center;         по горизонтали
align-items: center;             по вертикали
gap: 20px;                       отступ между элементами
flex-direction: column;          в столбик

position: relative;              для родителя
position: absolute;              поверх, точное место
position: fixed;                 прилипнуть к экрану
top / right / bottom / left: 10px;
z-index: 10;                     кто сверху

----------------------------------------
АНИМАЦИИ / ЭФФЕКТЫ
----------------------------------------
transition: all 0.3s ease;       плавность при hover
transform: scale(1.05);          увеличить
transform: translateY(-5px);     поднять вверх
transform: rotate(45deg);        повернуть

.element:hover { ... }           при наведении
.element:active { ... }          при нажатии
